API接入

秘塔AI提供API接口，支持程序化访问：

Python 
import requests

# 设置API密钥
API_KEY = "your-metaso-api-key"
API_URL = "https://api.metaso.cn/v1/search"

# 准备请求数据
headers = {
    "Authorization": f"Bearer {API_KEY}",
    "Content-Type": "application/json"
}

data = {
    "query": "劳动合同法第38条",
    "mode": "legal",  # legal/academic/web
    "top_k": 10
}

# 发送请求
response = requests.post(API_URL, headers=headers, json=data)
result = response.json()

# 输出结果
print(result['answer'])
print(f"参考资料: {len(result['references'])}条")
